// Title: Two Sum
            // Difficulty: Easy
            // Language: Python3

    def twoSum(self, nums: List[int], target: int) -> List[int]:
        for i in range(len(nums)):
            needed=target - nums[i]
        
        map={}
            map[nums[i]]=i
            if needed in map:
                return [map[needed],i]
