// Title: Subarray Sum Equals K
            // Difficulty: Medium
            // Language: Python3
            // Link: https://leetcode.com/problems/subarray-sum-equals-k/

class Solution:
    def subarraySum(self, nums: List[int], k: int) -> int:
        count=0
        maps={}
        maps[0]=1
        prefix_sum=0
        for i in range(len(nums)):
            prefix_sum+=nums[i]
            
            if prefix_sum - k in maps:
                count+=maps[prefix_sum-k]
            maps[prefix_sum]=maps.get(prefix_sum,0)+1
            #same meaning as

            #if prefix_sum in mp:
            #   maps[prefix_sum] += 1
            #else:
            #   maps[prefix_sum] =1
        return count
